### /lib/auth/session.ts

```typescript
import { SignJWT, jwtVerify } from 'jose';

const ACCESS_SECRET = new TextEncoder().encode(process.env.JWT_ACCESS_SECRET!);
const REFRESH_SECRET = new TextEncoder().encode(process.env.JWT_REFRESH_SECRET!);

export interface TokenPayload {
  sub: string;
  email: string;
  name: string;
  role: 'admin' | 'editor' | 'author';
  jti: string;
}

export async function createAccessToken(payload: Omit<TokenPayload, 'jti'> & { jti: string }): Promise<string> {
  return new SignJWT(payload)
    .setProtectedHeader({ alg: 'HS256' })
    .setIssuedAt()
    .setExpirationTime('15m')
    .sign(ACCESS_SECRET);
}

export async function verifyAccessToken(token: string): Promise<TokenPayload> {
  const { payload } = await jwtVerify(token, ACCESS_SECRET, { clockTolerance: 60 });
  return payload as TokenPayload;
}

export async function setAuthCookies(access: string, refresh: string) {
  const { cookies } = await import('next/headers');
  const cookieStore = await cookies();
  cookieStore.set('access_token', access, {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'strict',
    path: '/',
    maxAge: 60 * 15,
  });
  cookieStore.set('refresh_token', refresh, {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'strict',
    path: '/',
    maxAge: 60 * 60 * 24 * 7,
  });
}
```

### /middleware.ts

```typescript
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { verifyAccessToken } from './lib/auth/session';

const ADMIN_PATH = process.env.ADMIN_PATH || '/admin';

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  
  // Public API routes
  if (pathname === '/api/health' || pathname === '/api/auth/login') {
    return NextResponse.next();
  }
  
  // Protect API routes
  if (pathname.startsWith('/api/')) {
    const accessToken = request.cookies.get('access_token')?.value;
    if (!accessToken) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    try {
      await verifyAccessToken(accessToken);
      return NextResponse.next();
    } catch {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
  }
  
  // Public files
  if (/\.(.*)$/.test(pathname)) {
    return NextResponse.next();
  }
  
  // Admin login is public
  if (pathname === `${ADMIN_PATH}/login`) {
    return NextResponse.next();
  }
  
  // All other admin routes need auth
  if (pathname.startsWith(ADMIN_PATH)) {
    const accessToken = request.cookies.get('access_token')?.value;
    if (!accessToken) {
      return NextResponse.redirect(new URL(`${ADMIN_PATH}/login`, request.url));
    }
    try {
      await verifyAccessToken(accessToken);
      return NextResponse.next();
    } catch {
      return NextResponse.redirect(new URL(`${ADMIN_PATH}/login`, request.url));
    }
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
};
```

### /app/(admin-public)/admin/login/page.tsx

```typescript
'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';

const ADMIN_PATH = process.env.NEXT_PUBLIC_ADMIN_PATH || '/admin';

export default function LoginPage() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);
  const router = useRouter();

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setLoading(true);
    setError('');

    try {
      const res = await fetch('/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
      });

      const data = await res.json();
      if (!data.success) throw new Error(data.error?.message || 'Invalid');

      router.push(ADMIN_PATH);
      router.refresh();
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed');
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="min-h-screen flex items-center justify-center">
      <form onSubmit={handleSubmit} className="w-full max-w-md p-8 space-y-4">
        <h1 className="text-2xl font-bold text-[var(--admin-text)]">تسجيل الدخول</h1>
        {error && <div className="p-3 rounded bg-red-500/10 text-red-400 text-sm">{error}</div>}
        <input type="email" value={email} onChange={e => setEmail(e.target.value)} className="admin-input" placeholder="Email" required />
        <input type="password" value={password} onChange={e => setPassword(e.target.value)} className="admin-input" placeholder="Password" required />
        <button type="submit" disabled={loading} className="admin-btn-primary w-full">{loading ? '...' : 'Login'}</button>
      </form>
    </div>
  );
}
```

### /app/api/auth/login/route.ts

```typescript
import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { users } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import { verifyPassword } from '@/lib/auth/password';
import { createAccessToken, setAuthCookies } from '@/lib/auth/session';

export async function POST(request: Request) {
  try {
    const body = await request.json();
    const user = await db.select().from(users).where(eq(users.email, body.email)).limit(1);
    
    if (!user.length) {
      return NextResponse.json({ success: false, error: { message: 'Invalid' } }, { status: 401 });
    }

    const valid = await verifyPassword(user[0].passwordHash, body.password);
    if (!valid) {
      return NextResponse.json({ success: false, error: { message: 'Invalid' } }, { status: 401 });
    }

    const token = await createAccessToken({
      sub: user[0].id,
      email: user[0].email,
      name: user[0].name,
      role: user[0].role,
      jti: crypto.randomUUID(),
    });

    await setAuthCookies(token, 'refresh');

    return NextResponse.json({ success: true, data: { user: { id: user[0].id, name: user[0].name } } });
  } catch {
    return NextResponse.json({ success: false, error: { message: 'Failed' } }, { status: 500 });
  }
}
```
