
## THE BUG

```
app/(admin)/layout.tsx        ← guards with auth check
├── admin/login/page.tsx      ← Client Component, needs NO auth
└── admin/page.tsx            ← Server Component, needs auth

Result: /admin/login → layout checks cookie → no cookie → redirect /admin/login → infinite loop
```

## THE FIX

Split into two route groups:

```
app/
├── (admin-guarded)/          ← Has auth layout
│   └── admin/
│       ├── page.tsx            ← Dashboard
│       ├── content/
│       ├── media/
│       └── settings/
│
└── (admin-public)/           ← NO auth layout
    └── admin/
        └── login/
            └── page.tsx        ← Login page
```

// /app/(admin-guarded)/layout.tsx
// Auth check for ALL routes in this group

import { redirect } from 'next/navigation';
import { cookies } from 'next/headers';
import { verifyAccessToken } from '@/lib/auth/session';
import { AdminShell } from '@/components/admin/shell';
import { getSettings } from '@/lib/db/queries';

const ADMIN_PATH = process.env.ADMIN_PATH || '/admin';

export default async function GuardedAdminLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  const cookieStore = await cookies();
  const accessToken = cookieStore.get('access_token')?.value;

  if (!accessToken) {
    redirect(`${ADMIN_PATH}/login`);
  }

  try {
    const payload = await verifyAccessToken(accessToken);
    const settings = await getSettings();
    
    return (
      <AdminShell 
        user={{
          name: payload.name,
          email: payload.email,
          role: payload.role,
        }}
        siteName={settings?.siteName || 'New Aeon'}
      >
        {children}
      </AdminShell>
    );
  } catch {
    redirect(`${ADMIN_PATH}/login`);
  }
}

### /middleware.ts (UPDATED — handle both route groups)

```typescript
// /middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { verifyAccessToken } from './lib/auth/session';

const ADMIN_PATH = process.env.ADMIN_PATH || '/admin';
const PUBLIC_FILE = /\.(.*)$/;

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  
  // Public API routes
  const publicApiRoutes = ['/api/health', '/api/auth/login'];
  if (publicApiRoutes.some(route => pathname.startsWith(route))) {
    return NextResponse.next();
  }
  
  // Protect API routes
  if (pathname.startsWith('/api/')) {
    const accessToken = request.cookies.get('access_token')?.value;
    
    if (!accessToken) {
      return NextResponse.json({ success: false, error: 'Unauthorized' }, { status: 401 });
    }
    
    try {
      await verifyAccessToken(accessToken);
      return NextResponse.next();
    } catch {
      return NextResponse.json({ success: false, error: 'Unauthorized' }, { status: 401 });
    }
  }
  
  if (PUBLIC_FILE.test(pathname)) {
    return NextResponse.next();
  }
  
  // Admin login page is public — don't redirect
  if (pathname === `${ADMIN_PATH}/login`) {
    return NextResponse.next();
  }
  
  // All other admin routes need auth
  if (pathname.startsWith(ADMIN_PATH)) {
    const accessToken = request.cookies.get('access_token')?.value;
    
    if (!accessToken) {
      return NextResponse.redirect(new URL(`${ADMIN_PATH}/login`, request.url));
    }
    
    try {
      const payload = await verifyAccessToken(accessToken);
      
      const requestHeaders = new Headers(request.headers);
      requestHeaders.set('x-user-id', payload.sub);
      requestHeaders.set('x-user-role', payload.role);
      
      return NextResponse.next({
        request: { headers: requestHeaders },
      });
    } catch {
      return NextResponse.redirect(new URL(`${ADMIN_PATH}/login`, request.url));
    }
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
};
```
