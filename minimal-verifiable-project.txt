### lib/db/schema.ts

```typescript
import { pgTable, uuid, varchar, text, jsonb, pgEnum } from 'drizzle-orm/pg-core';
import type { ContentBlock } from '../types/content-blocks';

export const contentStatusEnum = pgEnum('content_status', ['draft', 'published', 'archived']);
export const localeEnum = pgEnum('locale', ['ar', 'en']);

export const content = pgTable('content', {
  id: uuid('id').primaryKey().defaultRandom(),
  slug: varchar('slug', { length: 255 }).notNull(),
  status: contentStatusEnum('status').default('draft'),
});

export const contentI18n = pgTable('content_i18n', {
  id: uuid('id').primaryKey().defaultRandom(),
  contentId: uuid('content_id').notNull().references(() => content.id, { onDelete: 'cascade' }),
  locale: localeEnum('locale').notNull(),
  title: varchar('title', { length: 255 }).notNull(),
  body: jsonb('body').$type<ContentBlock[]>(),
});
```

import { hash, verify, argon2id } from 'argon2';

export async function hashPassword(plain: string): Promise<string> {
  return hash(plain, { type: argon2id, memoryCost: 65536, timeCost: 3, parallelism: 4 });
}

export async function verifyPassword(hash: string, plain: string): Promise<boolean> {
  return verify(hash, plain);
}

### lib/auth/session.ts

```typescript
import { SignJWT, jwtVerify } from 'jose';

const ACCESS_SECRET = new TextEncoder().encode(process.env.JWT_ACCESS_SECRET!);

export async function createAccessToken(payload: { sub: string; email: string; name: string; role: string; jti: string }): Promise<string> {
  return new SignJWT(payload).setProtectedHeader({ alg: 'HS256' }).setIssuedAt().setExpirationTime('15m').sign(ACCESS_SECRET);
}

export async function verifyAccessToken(token: string): Promise<any> {
  const { payload } = await jwtVerify(token, ACCESS_SECRET, { clockTolerance: 60 });
  return payload;
}

export async function setAuthCookies(access: string, refresh: string) {
  const { cookies } = await import('next/headers');
  const cookieStore = await cookies();
  cookieStore.set('access_token', access, { httpOnly: true, path: '/', maxAge: 60 * 15 });
}
```

### middleware.ts

```typescript
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { verifyAccessToken } from './lib/auth/session';

const ADMIN_PATH = '/admin';

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  
  if (pathname === '/api/auth/login') return NextResponse.next();
  
  if (pathname.startsWith('/api/')) {
    const token = request.cookies.get('access_token')?.value;
    if (!token) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    try { await verifyAccessToken(token); return NextResponse.next(); } 
    catch { return NextResponse.json({ error: 'Unauthorized' }, { status: 401 }); }
  }
  
  if (pathname === `${ADMIN_PATH}/login`) return NextResponse.next();
  
  if (pathname.startsWith(ADMIN_PATH)) {
    const token = request.cookies.get('access_token')?.value;
    if (!token) return NextResponse.redirect(new URL(`${ADMIN_PATH}/login`, request.url));
    try { await verifyAccessToken(token); return NextResponse.next(); } 
    catch { return NextResponse.redirect(new URL(`${ADMIN_PATH}/login`, request.url)); }
  }
  
  return NextResponse.next();
}

export const config = { matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'] };
```
