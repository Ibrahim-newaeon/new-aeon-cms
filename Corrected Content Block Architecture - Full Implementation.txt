// Canonical ContentBlock union — our format, not TipTap's

export type ContentBlock =
  // Text (simple, no marks — for hero subtitles, captions, etc.)
  | { type: 'heading'; level: 1 | 2 | 3 | 4; text: string; anchor?: string }
  | { type: 'paragraph'; text: string; align?: 'left' | 'center' | 'right' | 'justify' }
  
  // Media
  | { type: 'image'; src: string; alt: string; caption?: string; width?: number; height?: number; layout: 'full' | 'wide' | 'normal' }
  | { type: 'gallery'; images: { src: string; alt: string }[]; layout: 'grid' | 'masonry' | 'carousel' | 'slideshow' }
  | { type: 'video'; url: string; provider: 'youtube' | 'vimeo' | 'self'; poster?: string; autoplay?: boolean }
  
  // Quotes
  | { type: 'quote'; text: string; author?: string; source?: string; style: 'bordered' | 'pull' }
  
  // Embeds

  // Spacing
  | { type: 'divider'; style: 'line' | 'space' | 'dots' | 'stars' }
  | { type: 'spacer'; height: number }
  
  // Raw HTML (sanitized)
  | { type: 'html'; content: string }

  | { type: 'table'; rows: number; cols: number; data: string[][]; headerRow?: boolean }
  | { type: 'accordion'; items: { title: string; content: ContentBlock[] }[] }

import { useState } from 'react';

export function BlockBuilder({ blocks, onChange }: BlockBuilderProps) {
  const [expandedIndex, setExpandedIndex] = useState<number | null>(null);
  const [showAddMenu, setShowAddMenu] = useState(false);

  const addBlock = (type: string) => {
    const newBlock = createDefaultBlock(type);
    onChange([...blocks, newBlock]);
    setExpandedIndex(blocks.length);
    setShowAddMenu(false);
  };

  const updateBlock = (index: number, block: ContentBlock) => {
    const updated = [...blocks];
    updated[index] = block;
    onChange(updated);
  };

  const removeBlock = (index: number) => {
    const updated = blocks.filter((_, i) => i !== index);
    onChange(updated);
  };

  const moveBlock = (from: number, to: number) => {
    const updated = [...blocks];
    const [moved] = updated.splice(from, 1);
    updated.splice(to, 0, moved);
    onChange(updated);
  };

  return (
    <div className="space-y-4">
      {blocks.map((block, idx) => (
        <BlockItem
          key={idx}
          index={idx}
          block={block}
          isExpanded={expandedIndex === idx}
          onToggle={() => setExpandedIndex(expandedIndex === idx ? null : idx)}
          onUpdate={(b) => updateBlock(idx, b)}
          onRemove={() => removeBlock(idx)}
          onMoveUp={() => moveBlock(idx, idx - 1)}
          onMoveDown={() => moveBlock(idx, idx + 1)}
          isFirst={idx === 0}
          isLast={idx === blocks.length - 1}
        />
      ))}

      <div className="relative">
        <button
          onClick={() => setShowAddMenu(!showAddMenu)}
          className="admin-btn-primary w-full"
        >
          <Plus size={18} />
          إضافة قسم
        </button>

        {showAddMenu && (
          <div className="absolute z-50 mt-2 w-full bg-[var(--admin-elevated)] border border-[var(--admin-line)] rounded-lg shadow-xl max-h-80 overflow-y-auto">
            <div className="p-2 grid grid-cols-2 gap-1">
              {Object.entries(blockTypeLabels).map(([type, label]) => (
                <button
                  key={type}
                  onClick={() => addBlock(type)}
                  className="text-start px-3 py-2 text-sm rounded hover:bg-white/5 transition-colors"
                >
                  {label}
                </button>
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

function createDefaultBlock(type: string): ContentBlock {
  switch (type) {
    case 'heading':
      return { type: 'heading', level: 2, text: '' };
    case 'paragraph':
      return { type: 'paragraph', text: '' };
    case 'image':
      return { type: 'image', src: '', alt: '', layout: 'normal' };
    case 'spacer':
      return { type: 'spacer', height: 2 };
    case 'divider':
      return { type: 'divider', style: 'line' };
    case 'rich-text':
      return { type: 'rich-text', content: { type: 'doc', content: [{ type: 'paragraph', content: [] }] } };
    case 'quote':
      return { type: 'quote', text: '', style: 'bordered' };
    case 'button':
      return { type: 'button', text: '', url: '', variant: 'primary', size: 'md' };
    case 'cta':
      return { type: 'cta', title: '', text: '', button: { text: '', url: '' } };
    default:
      return { type: type as any, text: '' };
  }
}

// /components/admin/rich-text-block-editor.tsx
'use client';

import { useEditor, EditorContent } from '@tiptap/react';
import StarterKit from '@tiptap/starter-kit';
import Image from '@tiptap/extension-image';
import Link from '@tiptap/extension-link';
import Placeholder from '@tiptap/extension-placeholder';
import { Bold, Italic, List, ListOrdered, Image as ImageIcon, Link as LinkIcon, Undo, Redo } from 'lucide-react';
import { cn } from '@/lib/utils';

interface RichTextBlockEditorProps {
  block: { type: 'rich-text'; content: object };
  onChange: (block: { type: 'rich-text'; content: object }) => void;
}

export function RichTextBlockEditor({ block, onChange }: RichTextBlockEditorProps) {
  const editor = useEditor({
    extensions: [
      StarterKit,
      Image,
      Link.configure({ openOnClick: false }),
      Placeholder.configure({ placeholder: 'ابدأ الكتابة هنا...' }),
    ],
    content: block.content,
    onUpdate: ({ editor }) => {
      onChange({
        type: 'rich-text',
        content: editor.getJSON(),
      });
    },
  });

  if (!editor) return null;

  return (
    <div className="border border-[var(--admin-line)] rounded-lg overflow-hidden">
      {/* Toolbar */}
      <div className="flex items-center gap-1 p-2 border-b border-[var(--admin-line)] bg-[var(--admin-elevated)] flex-wrap">
        <ToolbarButton onClick={() => editor.chain().focus().toggleBold().run()} isActive={editor.isActive('bold')} title="Bold">
          <Bold size={16} />
        </ToolbarButton>
        <ToolbarButton onClick={() => editor.chain().focus().toggleItalic().run()} isActive={editor.isActive('italic')} title="Italic">
          <Italic size={16} />
        </ToolbarButton>
        <div className="w-px h-6 bg-[var(--admin-line)] mx-1" />
        <ToolbarButton onClick={() => editor.chain().focus().toggleBulletList().run()} isActive={editor.isActive('bulletList')} title="Bullet List">
          <List size={16} />
        </ToolbarButton>
        <ToolbarButton onClick={() => editor.chain().focus().toggleOrderedList().run()} isActive={editor.isActive('orderedList')} title="Numbered List">
          <ListOrdered size={16} />
        </ToolbarButton>
        <div className="w-px h-6 bg-[var(--admin-line)] mx-1" />
        <ToolbarButton onClick={() => {
          const url = window.prompt('رابط الصورة:');
          if (url) editor.chain().focus().setImage({ src: url }).run();
        }} title="Image">
          <ImageIcon size={16} />
        </ToolbarButton>
        <ToolbarButton onClick={() => {
          const url = window.prompt('الرابط:');
          if (url) editor.chain().focus().setLink({ href: url }).run();
        }} title="Link">
          <LinkIcon size={16} />
        </ToolbarButton>
        <div className="w-px h-6 bg-[var(--admin-line)] mx-1" />
        <ToolbarButton onClick={() => editor.chain().focus().undo().run()} title="Undo">
          <Undo size={16} />
        </ToolbarButton>
        <ToolbarButton onClick={() => editor.chain().focus().redo().run()} title="Redo">
          <Redo size={16} />
        </ToolbarButton>
      </div>
      
      {/* Editor */}
      <EditorContent
        editor={editor}
        className="prose prose-invert max-w-none p-4 min-h-[300px] bg-[var(--admin-surface)]"
      />
    </div>
  );
}

function ToolbarButton({ onClick, isActive, children, title }: { onClick: () => void; isActive?: boolean; children: React.ReactNode; title: string }) {
  return (
    <button
      type="button"
      onClick={onClick}
      title={title}
      className={cn(
        "p-2 rounded hover:bg-white/5 transition-colors",
        isActive && "bg-[var(--admin-primary-muted)] text-[var(--admin-primary)]"
      )}
    >
      {children}
    </button>
  );
}

// /components/site/content-renderer.tsx
import { generateHTML } from '@tiptap/html';
import StarterKit from '@tiptap/starter-kit';
import Image from '@tiptap/extension-image';
import Link from '@tiptap/extension-link';
import type { ContentBlock } from '@/lib/types/content-blocks';

// TipTap extensions for HTML generation
const tiptapExtensions = [StarterKit, Image, Link];

interface ContentRendererProps {
  blocks: ContentBlock[] | null | undefined;
}

export function ContentRenderer({ blocks }: ContentRendererProps) {
  if (!blocks || !Array.isArray(blocks)) return null;

  return (
    <div className="space-y-6">
      {blocks.map((block, idx) => (
        <BlockRenderer key={idx} block={block} />
      ))}
    </div>
  );
}

function BlockRenderer({ block }: { block: ContentBlock }) {
  switch (block.type) {
    // --- Simple text blocks ---
    
    case 'heading':
      const HeadingTag = `h${block.level}` as keyof JSX.IntrinsicElements;
      return (
        <HeadingTag 
          id={block.anchor} 
          className={`font-bold text-gray-900 ${
            block.level === 1 ? 'text-3xl' : 
            block.level === 2 ? 'text-2xl' : 
            block.level === 3 ? 'text-xl' : 'text-lg'
          }`}
        >
          {block.text}
        </HeadingTag>
      );

    case 'paragraph':
      return (
        <p className={cn("text-gray-700 leading-relaxed", {
          'text-right': block.align === 'right' || !block.align,
          'text-left': block.align === 'left',
          'text-center': block.align === 'center',
          'text-justify': block.align === 'justify',
        })}>
          {block.text}
        </p>
      );

    // --- Media blocks ---
    
    case 'image':
      return (
        <figure className={cn("my-8", {
          'max-w-full': block.layout === 'full',
          'max-w-4xl mx-auto': block.layout === 'wide',
          'max-w-2xl mx-auto': block.layout === 'normal',
        })}>
          <img 
            src={block.src} 
            alt={block.alt} 
            className="w-full rounded-lg"
            width={block.width}
            height={block.height}
          />
          {block.caption && (
            <figcaption className="text-center text-sm text-gray-500 mt-2">
              {block.caption}
            </figcaption>
          )}
        </figure>
      );

    case 'gallery':
      return (
        <div className={cn("grid gap-4", {
          'grid-cols-2 md:grid-cols-3': block.layout === 'grid',
          'columns-2 md:columns-3': block.layout === 'masonry',
        })}>
          {block.images.map((img, idx) => (
            <img 
              key={idx}
              src={img.src} 
              alt={img.alt} 
              className="w-full rounded-lg object-cover break-inside-avoid"
            />
          ))}
        </div>
      );

    case 'video':
      if (block.provider === 'youtube') {
        const videoId = extractYouTubeId(block.url);
        return (
          <div className="aspect-video">
            <iframe
              src={`https://www.youtube.com/embed/${videoId}${block.autoplay ? '?autoplay=1' : ''}`}
              className="w-full h-full rounded-lg"
              allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
              allowFullScreen
            />
          </div>
        );
      }
      return <a href={block.url} className="text-blue-600 underline">{block.url}</a>;

    // --- Quote ---
    
    case 'quote':
      return (
        <blockquote className={cn("pr-4 py-2 my-6 rounded-r-lg", {
          'border-r-4 border-indigo-500 bg-gray-50': block.style === 'bordered',
          'text-xl italic': block.style === 'pull',
        })}>
          <p className="text-lg text-gray-700 italic">{block.text}</p>
          {block.author && (
            <cite className="block mt-2 text-sm text-gray-500 not-italic">
              — {block.author}
              {block.source && `, ${block.source}`}
            </cite>
          )}
        </blockquote>
      );

    // --- Embed ---
    
    case 'embed':
      return (
        <div className="my-6">
          <a href={block.url} target="_blank" rel="noopener noreferrer" className="text-blue-600 underline">
            {block.type === 'instagram' && 'Instagram'}
            {block.type === 'twitter' && 'Twitter'}
            {block.type === 'tiktok' && 'TikTok'}
            {block.type === 'facebook' && 'Facebook'}
            : {block.url}
          </a>
        </div>
      );

    // --- Button ---
    
    case 'button':
      return (
        <a
          href={block.url}
          className={cn("inline-flex items-center px-6 py-3 rounded-lg font-medium transition-colors", {
            'bg-indigo-600 text-white hover:bg-indigo-700': block.variant === 'primary',
            'bg-gray-200 text-gray-900 hover:bg-gray-300': block.variant === 'secondary',
            'border-2 border-indigo-600 text-indigo-600 hover:bg-indigo-50': block.variant === 'outline',
            'text-indigo-600 hover:bg-indigo-50': block.variant === 'ghost',
            'text-sm px-4 py-2': block.size === 'sm',
            'px-6 py-3': block.size === 'md',
            'text-lg px-8 py-4': block.size === 'lg',
            'w-full justify-center': block.fullWidth,
          })}
        >
          {block.text}
        </a>
      );

    // --- Spacing ---
    
    case 'divider':
      if (block.style === 'space') {
        return <div className="py-8" />;
      }
      return <hr className={cn("my-8", {
        'border-gray-200': block.style === 'line',
        'border-dotted border-gray-300': block.style === 'dots',
        'border-gray-200 opacity-50': block.style === 'stars',
      })} />;

    case 'spacer':
      return <div style={{ height: `${block.height}rem` }} />;

    // --- Raw HTML ---
    
    case 'html':
      return <div dangerouslySetInnerHTML={{ __html: block.content }} />;

    // --- Complex layout: CTA ---
    
    case 'cta':
      return (
        <section className={cn("relative py-16 px-4 rounded-xl overflow-hidden", {
          'bg-gradient-to-br from-indigo-900 to-purple-900': !block.backgroundImage,
        })}>
          {block.backgroundImage && (
            <>
              <img src={block.backgroundImage} alt="" className="absolute inset-0 w-full h-full object-cover" />
              {block.overlay && <div className="absolute inset-0 bg-black/40" />}
            </>
          )}
          <div className="relative z-10 text-center max-w-2xl mx-auto">
            <h2 className="text-3xl font-bold text-white mb-4">{block.title}</h2>
            <p className="text-lg text-white/90 mb-8">{block.text}</p>
            <a
              href={block.button.url}
              className="inline-flex items-center px-8 py-3 bg-white text-gray-900 rounded-full font-medium hover:bg-gray-100 transition-colors"
            >
              {block.button.text}
            </a>
          </div>
        </section>
      );

    // --- Complex layout: Feature Grid ---
    
    case 'feature-grid':
      return (
        <div className={cn("grid gap-6", {
          'grid-cols-2': block.columns === 2,
          'grid-cols-3': block.columns === 3,
          'grid-cols-4': block.columns === 4,
        })}>
          {block.items.map((item, idx) => (
            <div key={idx} className="p-6 rounded-lg bg-gray-50">
              {item.icon && <div className="text-3xl mb-4">{item.icon}</div>}
              <h3 className="font-semibold text-lg mb-2">{item.title}</h3>
              <p className="text-gray-600">{item.description}</p>
            </div>
          ))}
        </div>
      );

    // --- Complex layout: Testimonial ---
    
    case 'testimonial':
      return (
        <div className="space-y-6">
          {block.items.map((item, idx) => (
            <div key={idx} className="p-6 rounded-lg bg-gray-50">
              <div className="flex gap-1 mb-4">
                {Array.from({ length: item.rating || 5 }).map((_, i) => (
                  <span key={i} className="text-yellow-400">★</span>
                ))}
              </div>
              <p className="text-lg italic mb-4">"{item.quote}"</p>
              <div className="flex items-center gap-3">
                {item.avatar && <img src={item.avatar} alt="" className="w-10 h-10 rounded-full" />}
                <div>
                  <p className="font-medium">{item.author}</p>
                  {item.role && <p className="text-sm text-gray-500">{item.role}</p>}
                </div>
              </div>
            </div>
          ))}
        </div>
      );

    // --- THE KEY CASE: Rich text via TipTap ---
    
    case 'rich-text': {
      // Generate HTML from TipTap document using TipTap's own renderer
      const html = generateHTML(block.content, tiptapExtensions);
      return (
        <div 
          className="prose prose-lg max-w-none"
          dangerouslySetInnerHTML={{ __html: html }}
        />
      );
    }

    // --- Fallback ---
    
    default:
      return (
        <div className="p-4 bg-yellow-50 text-yellow-800 rounded-lg">
          نوع القسم غير معروف: {(block as any).type}
        </div>
      );
  }
}

// Helper functions
function cn(...classes: (string | false | undefined | Record<string, boolean>)[]) {
  return classes
    .flatMap(c => {
      if (typeof c === 'string') return c;
      if (typeof c === 'object' && c !== null) {
        return Object.entries(c).filter(([, v]) => v).map(([k]) => k);
      }
      return [];
    })
    .filter(Boolean)
    .join(' ');
}

function extractYouTubeId(url: string): string {
  const match = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/|youtube\.com\/embed\/)([^&\s]+)/);
  return match?.[1] || '';
}

              placeholder="about-us"
            />
          </div>
          
          <div>
            <label className="block text-sm mb-2">الحالة</label>
            <select
              value={formData.status}
              onChange={(e) => setFormData(prev => ({ ...prev, status: e.target.value as any }))}
              className="admin-input"
            >
              <option value="draft">مسودة</option>
              <option value="published">منشور</option>
              <option value="archived">مؤرشف</option>
            </select>
          </div>
        </div>

        <div>

          <input
            type="url"
            value={formData.featuredImage}
            onChange={(e) => setFormData(prev => ({ ...prev, featuredImage: e.target.value }))}
            className="admin-input"
            placeholder="https://example.com/image.jpg"
          />
        </div>

        {(locale) => (
          <div className="space-y-4">
            <div>
              <label className="block text-sm mb-2">العنوان *</label>
              <input
                type="text"
                value={formData.translations[locale].title}

                className="admin-input"
                placeholder="عنوان الصفحة"
              />
            </div>

            <div>
              <label className="block text-sm mb-2">مقتطف</label>
              <textarea
                value={formData.translations[locale].excerpt}
                onChange={(e) => updateTranslation(locale, 'excerpt', e.target.value)}

                placeholder="وصف مختصر للصفحة"

              onChange={(values) => {
                setFormData(prev => ({
                  ...prev,
                  translations: {
                    ...prev.translations,
                    [locale]: {
                      ...prev.translations[locale],
                      ...values,
                    },
                  },
                }));
              }}
            />
          </div>
        )}
      </LocaleTabs>

      <div className="flex items-center gap-3">
        <button
          onClick={handleSubmit}
          disabled={loading}
          className="admin-btn-primary disabled:opacity-50"
        >
          {loading ? 'جاري الحفظ...' : 'حفظ الصفحة'}
        </button>
        <button
          type="button"
          onClick={() => router.back()}
          className="admin-btn-ghost"
        >
          إلغاء
        </button>
      </div>
    </div>
  );
}
